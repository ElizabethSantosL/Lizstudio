        // Category Filter Controller
        function filterCategory(cat) {
            currentCategory = cat;
            
            // Update active button state
            document.querySelectorAll('.cat-btn').forEach(btn => {
                btn.className = 'cat-btn px-4 py-2 rounded-full text-xs sm:text-sm font-medium tracking-wide transition-all bg-white text-taupe hover:bg-gold-light/30 border border-gold-light/60';
            });

            const activeBtn = document.getElementById(`btn-${cat}`);
            if (activeBtn) {
                activeBtn.className = 'cat-btn px-4 py-2 rounded-full text-xs sm:text-sm font-medium tracking-wide transition-all bg-pine text-cream shadow-sm';
            }

            // Update title
            const titles = {
                all: 'Catálogo Completo',
                confeitaria: '🎂 Confeitaria & Cenografia',
                props: '📱 Props Realistas',
                religiosa: '🕊️ Linha Religiosa'
            };
            document.getElementById('current-category-title').innerText = titles[cat];

            renderProducts();
        }

            searchQuery = document.getElementById('search-input').value;

            document.getElementById('search-input').value = '';

        // Cart Drawer Handlers
        function toggleCartDrawer() {
            const drawer = document.getElementById('cart-drawer');
            const backdrop = document.getElementById('cart-backdrop');
            const panel = document.getElementById('cart-panel');

            if (panel.classList.contains('translate-x-full')) {
                // Open
                drawer.classList.remove('pointer-events-none');
                backdrop.classList.remove('opacity-0', 'pointer-events-none');
                backdrop.classList.add('opacity-100');
                panel.classList.remove('translate-x-full');
            } else {
                // Close
                panel.classList.add('translate-x-full');
                backdrop.classList.remove('opacity-100');
                backdrop.classList.add('opacity-0', 'pointer-events-none');
                setTimeout(() => {
                    drawer.classList.add('pointer-events-none');
                }, 300);
            }
        }

        // Add Product to Cart
        function addToCart(productId) {
            const product = productsData.find(p => p.id === productId);
            if (!product) return;

            const existing = cart.find(item => item.id === productId);
            if (existing) {
                existing.quantity += 1;
            } else {
                cart.push({ ...product, quantity: 1 });
            }

            updateCartUI();
            
            // Subtle notification animation on cart button
            const badge = document.getElementById('cart-badge');
            badge.classList.add('scale-125');
            setTimeout(() => badge.classList.remove('scale-125'), 200);
        }

        // Update Item Quantity in Cart
        function updateQuantity(productId, delta) {
            const item = cart.find(i => i.id === productId);
            if (!item) return;

            item.quantity += delta;
            if (item.quantity <= 0) {
                cart = cart.filter(i => i.id !== productId);
            }

            updateCartUI();
        }

        // Remove Item Completely
        function removeFromCart(productId) {
            cart = cart.filter(i => i.id !== productId);
            updateCartUI();
        }

        // Update Cart Drawer UI
        function updateCartUI() {
            const container = document.getElementById('cart-items-container');
            const badge = document.getElementById('cart-badge');
            const totalElem = document.getElementById('cart-total');
            const subtotalElem = document.getElementById('cart-subtotal');

            const totalCount = cart.reduce((sum, item) => sum + item.quantity, 0);
            const totalPrice = cart.reduce((sum, item) => sum + (item.price * item.quantity), 0);

            badge.textContent = totalCount;
            totalElem.textContent = formatMoney(totalPrice);
            subtotalElem.textContent = formatMoney(totalPrice);

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 text-taupe-muted">
                        <div class="w-16 h-16 mx-auto mb-3 rounded-full bg-gold-light/30 flex items-center justify-center text-gold text-2xl">
                            <i class="fa-solid fa-basket-shopping"></i>
                        </div>
                        <p class="font-heading text-lg font-medium text-pine mb-1">Seu pedido está vazio</p>
                        <p class="text-xs">Navegue pelo catálogo e adicione suas peças preferidas!</p>
                    </div>
                `;
                return;
            }

            container.innerHTML = '';
            cart.forEach(item => {
                const fallbackImg = getSVGPlaceholder(item.name, item.category);
                const itemDiv = document.createElement('div');
                itemDiv.className = 'flex items-center gap-3 p-3 bg-white rounded-xl border border-gold-light/50 shadow-sm';
                
                itemDiv.innerHTML = `
                    <img src="${item.image}" onerror="this.src='${fallbackImg}'" alt="${item.name}" class="w-16 h-16 object-cover rounded-lg bg-cream-card">
                    
                    <div class="flex-grow">
                        <h4 class="font-heading font-semibold text-xs text-pine line-clamp-1">${item.name}</h4>
                        <span class="text-xs text-crimson font-bold block mb-1">${formatMoney(item.price)}</span>
                        
                        <!-- Quantity Controller -->
                        <div class="flex items-center gap-2">
                            <button onclick="updateQuantity('${item.id}', -1)" class="w-5 h-5 rounded bg-cream-card border border-gold-light/60 text-taupe flex items-center justify-center text-xs hover:bg-gold-light/30">
                                -
                            </button>
                            <span class="text-xs font-semibold px-1 text-taupe">${item.quantity}</span>
                            <button onclick="updateQuantity('${item.id}', 1)" class="w-5 h-5 rounded bg-cream-card border border-gold-light/60 text-taupe flex items-center justify-center text-xs hover:bg-gold-light/30">
                                +
                            </button>
                        </div>
                    </div>

                    <div class="text-right flex flex-col justify-between items-end h-16">
                        <button onclick="removeFromCart('${item.id}')" class="text-taupe-muted hover:text-crimson text-xs">
                            <i class="fa-regular fa-trash-can"></i>
                        </button>
                        <span class="font-bold text-xs text-pine font-heading">
                            ${formatMoney(item.price * item.quantity)}
                        </span>
                    </div>
                `;
                container.appendChild(itemDiv);
            });
        }

        // WhatsApp Checkout Order Link Generator
        function sendWhatsAppOrder() {
            if (cart.length === 0) {
                alert("Seu pedido está vazio! Escolha algum produto primeiro.");
                return;
            }

            const phone = "5585992060056";
            let message = "🎄 *LIZ STUDIO - NOVO PEDIDO DE NATAL* 🎄\n\n";
            message += "Olá! Gostaria de encomendar os seguintes itens do catálogo:\n\n";

            let total = 0;
            cart.forEach((item, index) => {
                const subtotal = item.price * item.quantity;
                total += subtotal;
                message += `${index + 1}. *${item.name}*\n   • Qtd: ${item.quantity}x | Valor: R$ ${subtotal.toFixed(2).replace('.', ',')}\n`;
            });

            message += `\n💰 *VALOR TOTAL ESTIMADO:* R$ ${total.toFixed(2).replace('.', ',')}\n`;
            message += "\nAguardo informações sobre prazos e envio. Obrigado!";

            const encodedMessage = encodeURIComponent(message);
            const whatsappUrl = `https://wa.me/${phone}?text=${encodedMessage}`;
            window.open(whatsappUrl, '_blank');
        }
